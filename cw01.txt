create table regions(region_id int primary key, region_name varchar2(50));

create table countries(country_id int primary key, region_id int, constraint fk_country_region foreign key (region_id) references regions(region_id));

create table locations(location_id int primary key, street_address varchar2(40), city varchar2(30) not null, state_province varchar2(25), country_id int);

create table departments(department_id int primary key, department_name varchar2(30) not null, manager_id int, location_id int, constraint fk_department_location foreign key (location_id) references locations(location_id));

create table jobs(job_id int primary key, job_title varchar2(35) not null, min_salary int, max_salary int);

create table employees(employee_id int primary key, first_name varchar2(20), last_name varchar2(25) not null, email varchar2(25) not null, phone_number varchar2(20), hire_date date not null, job_id int not null, salary int, commission_pct int, manager_id int, department_id int, constraint fk_employee_job foreign key (job_id) references jobs(job_id), constraint fk_employee_department foreign key (department_id) references departments(department_id), constraint fk_employee_manager foreign key (manager_id) references employees(employee_id));

create table job_history(employee_id int not null, start_date date not null, job_id int not null, department_id int, constraint pk_job_history primary key (employee_id, start_date), constraint fk_job_history_employee foreign key (employee_id) references employees(employee_id), constraint fk_job_history_job foreign key (job_id) references jobs(job_id), constraint fk_job_history_department foreign key (department_id) references departments(department_id));

alter table countries add country_name varchar2(40);

alter table locations add postal_code varchar2(12);

alter table job_history add end_date date;

alter table departments add constraint fk_department_manager foreign key (manager_id) references employees(employee_id);
